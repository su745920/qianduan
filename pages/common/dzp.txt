      getLuckyDraw().then((resData) => {
        if (resData.code === 4001) return uni.showModal({
          title: '提示',
          showCancel: false,

            if (res.confirm) {

              })
            }
          }
        });
        LuckyDrawData.value = [...resData.data]

          prizes.value.push({

            imgs: [{
              src: item.image as string,
              width: '25%',
              top: '45%',
              left: '-2%'
            }]
          },)
        })
        return prizes.value
      })
    }
  }
}
